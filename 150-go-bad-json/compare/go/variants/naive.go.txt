// Sign-ups for the bake sale: who's coming, and how many guests.
package main

import (
	"encoding/json"
	"log"
	"net/http"
	"sync"
)

type Signup struct {
	Name   string `json:"name"`
	Email  string `json:"email"`
	Guests int    `json:"guests"`
}

var (
	mu      sync.Mutex
	signups = []Signup{}
)

func main() {
	http.HandleFunc("POST /api/signups", func(w http.ResponseWriter, r *http.Request) {
		var s Signup
		if err := json.NewDecoder(r.Body).Decode(&s); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		mu.Lock()
		signups = append(signups, s)
		mu.Unlock()
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusCreated)
		json.NewEncoder(w).Encode(s)
	})
	http.HandleFunc("GET /api/signups", func(w http.ResponseWriter, r *http.Request) {
		mu.Lock()
		defer mu.Unlock()
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(signups)
	})
	log.Fatal(http.ListenAndServe(":8090", nil))
}
