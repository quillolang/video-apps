// The bakery's hit counter: every visit adds one, and /hits says how many.
package main

import (
	"fmt"
	"net/http"
)

var hits int

func main() {
	http.HandleFunc("GET /{$}", func(w http.ResponseWriter, r *http.Request) {
		hits++
		fmt.Fprintf(w, "Welcome! You're visitor number %d.\n", hits)
	})
	http.HandleFunc("GET /hits", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintf(w, "%d\n", hits)
	})
	http.ListenAndServe(":8090", nil)
}
